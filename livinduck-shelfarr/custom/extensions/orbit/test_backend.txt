result = nil
Book.transaction do
  Book.transaction(requires_new: true) do
    result=RequestCreationService.call(user:user,work_id:id,book_types:['audiobook'],metadata_attrs:attrs,language:'en')
    assert(search_jobs.call.empty?, 'search queued before inner transaction committed')
  end
  assert(search_jobs.call.empty?, 'search queued before outer transaction committed')
end
assert(search_jobs.call.size == 1, 'committed request must enqueue exactly one search')
assert(search_jobs.call.first[:args] == [result.created_requests.first.id], 'wrong request queued')
Book.transaction do
  SearchJob.perform_later(result.created_requests.first.id)
  raise ActiveRecord::Rollback
end
assert(search_jobs.call.size == 1, 'rolled-back transaction queued a search')
SearchJob.perform_later(result.created_requests.first.id)
assert(search_jobs.call.size == 2, 'search outside a transaction was not enqueued immediately')
puts 'RESULT '+{search_enqueue_after_commit:true,nested_transaction:true,rollback_discards_job:true,non_transaction_enqueue:true}.to_json
assert(result.success?,"creation failed: #{result.errors}")

duplicate=RequestCreationService.call(user:user,work_id:id,book_types:['audiobook'],metadata_attrs:attrs)
